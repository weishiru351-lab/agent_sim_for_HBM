cmake_minimum_required(VERSION 3.14)
project(HBMAddressBridge LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT RAMULATOR_SOURCE)
  message(FATAL_ERROR "Use prepare.py build to verify and configure the pinned sources")
endif()
set(RAMULATOR_PYTHON_BINDINGS OFF CACHE BOOL "No native Python binding required" FORCE)
add_subdirectory("${RAMULATOR_SOURCE}" "${CMAKE_BINARY_DIR}/ramulator")
add_executable(hbm_ramulator driver.cpp)
target_link_libraries(hbm_ramulator PRIVATE ramulator fmt::fmt)
target_compile_definitions(hbm_ramulator PRIVATE RAMULATOR_REVISION="72427a1bba3771564c4fb0e494ba02242fd1eaa7")
